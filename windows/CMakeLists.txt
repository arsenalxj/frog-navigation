cmake_minimum_required(VERSION 3.25)
project(Frog VERSION 1.0.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "请使用 Windows MSVC x64 工具链")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_compile_options(/W4 /permissive- /utf-8 /EHsc /Zc:__cplusplus)
add_compile_definitions(UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
set(JSON_HEADER "${CMAKE_BINARY_DIR}/_deps/json.hpp")
if(NOT EXISTS "${JSON_HEADER}")
  file(DOWNLOAD "https://raw.githubusercontent.com/nlohmann/json/v3.11.3/single_include/nlohmann/json.hpp"
       "${JSON_HEADER}" TLS_VERIFY ON STATUS DOWNLOAD_RESULT)
  list(GET DOWNLOAD_RESULT 0 DOWNLOAD_CODE)
  if(NOT DOWNLOAD_CODE EQUAL 0)
    message(FATAL_ERROR "下载 JSON 依赖失败：${DOWNLOAD_RESULT}")
  endif()
endif()
file(SHA256 "${JSON_HEADER}" JSON_HASH)
if(NOT JSON_HASH STREQUAL "9bea4c8066ef4a1c206b2be5a36302f8926f7fdc6087af5d20b417d0cf103ea6")
  message(FATAL_ERROR "JSON 依赖 SHA-256 不匹配，请删除 build/_deps/json.hpp 后重试")
endif()
add_library(frog_core STATIC src/core/Util.cpp src/core/Model.cpp src/core/Storage.cpp src/core/State.cpp)
target_include_directories(frog_core PUBLIC src "${CMAKE_BINARY_DIR}/_deps")
target_link_libraries(frog_core PUBLIC ole32 shell32 shlwapi bcrypt winhttp icu)
add_library(frog_platform STATIC src/platform/Integration.cpp src/platform/Images.cpp)
target_link_libraries(frog_platform PUBLIC frog_core windowscodecs d2d1 d3d11 comctl32 advapi32 user32 gdi32 oleacc)
add_executable(Frog WIN32 src/Main.cpp src/ui/Launcher.cpp src/ui/Panels.cpp src/ui/Accessibility.cpp resources/Frog.rc)
target_link_libraries(Frog PRIVATE frog_platform d2d1 dwrite dwmapi uxtheme imm32 uiautomationcore)
target_link_options(Frog PRIVATE /MANIFEST:NO)
include(CTest)
if(BUILD_TESTING)
  add_executable(FrogTests tests/CoreTests.cpp)
  target_link_libraries(FrogTests PRIVATE frog_platform)
  add_test(NAME core COMMAND FrogTests "${CMAKE_CURRENT_SOURCE_DIR}/tests/fixtures/macos-v1.json")
  add_executable(FrogImageTests tests/ImageTests.cpp)
  target_link_libraries(FrogImageTests PRIVATE frog_platform ws2_32)
  add_test(NAME image-cache COMMAND FrogImageTests)
  set_tests_properties(image-cache PROPERTIES TIMEOUT 30)
  add_executable(FrogSmoke tests/Smoke.cpp)
  target_link_libraries(FrogSmoke PRIVATE frog_platform uiautomationcore)
  add_test(NAME native-smoke COMMAND FrogSmoke "$<TARGET_FILE:Frog>" "${CMAKE_BINARY_DIR}/Testing/artifacts")
  set_tests_properties(native-smoke PROPERTIES TIMEOUT 60)
  add_executable(FrogIconProbe tests/IconProbe.cpp)
  target_link_libraries(FrogIconProbe PRIVATE frog_platform)
endif()
